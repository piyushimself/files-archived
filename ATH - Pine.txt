// v7 = v6 + CONFIRMATION-TRIGGER ENTRY (no more entry on the wick itself)
//
// Changed from v6:
//   - A month that makes a fresh ATH wick + green close (renamed
//     athSignal, was "monthlyBreakout") no longer enters a trade
//     by itself. It only ARMS a trigger level = that wick high
//     plus a small buffer ("just above the wick").
//   - Entry (BUY / ADD) only fires later, on the first CONFIRMED
//     month whose high trades above that armed trigger level.
//     Entry price, SL, and both targets are all built off that
//     trigger level - not the original signal candle's wick, and
//     not the confirming candle's close.
//   - The armed level (fuchsia) and the signal candle marker are
//     now plotted on the chart so you can see what's pending.
//   - Everything else (weekly exit, WATCH/READY, scoring,
//     pyramiding, capital sizing engine, dashboard layout) is
//     unchanged from v6.
// ============================================================

priorATH := runningATH

if barstate.isconfirmed
    if na(runningATH)

    else
        runningATH := math.max(runningATH, high)

// which closes green. This is the old entry condition - now it only
// arms a confirmation trigger (see below), it does not buy anything.
athSignal = barstate.isconfirmed and not na(priorATH) and high > priorATH and monthlyIsBullish

if barstate.isconfirmed

monthsSinceATH = consolMonths

// ============================================================
// CONFIRMATION TRIGGER — the actual entry mechanism.
// athSignal arms pendingLevel = signal wick high + buffer.
// Entry fires on the first LATER confirmed month whose high
// trades above pendingLevel. Entry price = pendingLevel itself
// (i.e. "just above the wick"), not the confirming candle's high
// or close.
// ============================================================
var float pendingLevel = na

confirmEntryTrigger = barstate.isconfirmed and not na(pendingLevel) and high > pendingLevel
triggerLevel = pendingLevel   // snapshot before this bar's arm/clear below

if barstate.isconfirmed
    if confirmEntryTrigger
        pendingLevel := na

volAvg       = ta.sma(volume, volAvgLen)

tightnessPct = (not na(consolHigh) and not na(consolLow) and consolHigh > 0) ? (consolHigh - consolLow) / consolHigh * 100 : na
isTightBase  = not na(tightnessPct) and tightnessPct <= tightPct

distanceFromATH = (not na(runningATH) and runningATH > 0) ? (runningATH - close) / runningATH * 100.0 : na

atrValue = ta.atr(atrLen)

smaFast = ta.sma(close, smaFastLen)
smaSlow = ta.sma(close, smaSlowLen)

higherHigh = high > high[1]
higherLow  = low > low[1]

risingLowsInBase = not athSignal and low > low[1]

volRecentAvg = ta.sma(volume, dryUpLookback)
volDryingUp  = volRecentAvg < volAvg

var float[] consolHistoryArr = array.new_float(0)

f_clamp01(x) =>
    math.max(0.0, math.min(1.0, x))

f_trendScore(c, smaF, smaS, hh, hl) =>
    float score = 0.0
    score += c > smaF ? 20.0 : 0.0
    score += smaF > smaS ? 20.0 : 0.0
    score += c > smaS ? 20.0 : 0.0
    score += hh ? 20.0 : 0.0
    score += hl ? 20.0 : 0.0
    score

f_consolLengthScore(months, historyArr, fbMed, fbHigh, staleMonths) =>
    float score = 0.0
    if array.size(historyArr) >= 2
        avgPast = array.sum(historyArr) / array.size(historyArr)
        ratio = avgPast > 0 ? months / avgPast : (months > 0 ? 2.0 : 0.0)
        if ratio >= 1.2
            score := 30.0
        else if ratio >= 0.7
            score := 20.0
        else if ratio >= 0.3
            score := 10.0
        else
            score := 0.0
    else
        if months >= fbHigh
            score := 30.0
        else if months >= fbMed
            score := 20.0
        else if months >= 1
            score := 10.0
        else
            score := 0.0
    if months >= staleMonths
        score := math.min(score, 20.0)
    score

f_baseScore(tightPctVal, tightThreshold, months, historyArr, fbMed, fbHigh, staleMonths, risingLows, dryUp) =>
    lengthScore = f_consolLengthScore(months, historyArr, fbMed, fbHigh, staleMonths)
    tightnessScore = na(tightPctVal) ? 0.0 : 30.0 * (1.0 - f_clamp01((tightPctVal - tightThreshold) / tightThreshold))
    risingScore = risingLows ? 20.0 : 0.0
    dryUpScore  = dryUp ? 20.0 : 0.0
    math.min(lengthScore + tightnessScore + risingScore + dryUpScore, 100.0)

f_volumeScore(volCur, volAvgVal, volMult, dryUp) =>
    expansionScore = volCur > volAvgVal * volMult ? 40.0 : (volCur > volAvgVal ? 20.0 : 0.0)
    aboveAvgScore  = volCur > volAvgVal ? 30.0 : 0.0
    dryUpScore     = dryUp ? 30.0 : 0.0
    math.min(expansionScore + aboveAvgScore + dryUpScore, 100.0)

getScoreColor(score) =>
    color result = color.orange
    if score >= 80
        result := color.lime
    else if score >= 60
        result := color.teal
    else if score >= 35
        result := color.yellow
    result

trendScoreLive = f_trendScore(close, smaFast, smaSlow, higherHigh, higherLow)
baseScoreLive  = f_baseScore(tightnessPct, tightPct, consolMonths, consolHistoryArr, fallbackMed, fallbackHigh, tooLongMonths, risingLowsInBase, volDryingUp)
volScoreLive   = f_volumeScore(volume, volAvg, volMultiplier, volDryingUp)
compositeScoreLive = (trendScoreLive + baseScoreLive + volScoreLive) / 3.0

// TRADE STATE — MULTIPLE OPEN LEGS (quantity + capital)
// ============================================================
var float[] legEntry     = array.new_float(0)
var float[] legSL        = array.new_float(0)
var float[] legT1        = array.new_float(0)
var float[] legT2        = array.new_float(0)
var int[]   legEntryBar  = array.new_int(0)
var float[] legEntryTime = array.new_float(0)
var float[] legRisk      = array.new_float(0)
var float[] legTrend     = array.new_float(0)
var float[] legBase      = array.new_float(0)
var float[] legVol       = array.new_float(0)
var int[]   legConsol    = array.new_int(0)
var float[] legQty       = array.new_float(0)   // sized quantity for this leg
var float[] legCapital   = array.new_float(0)   // capital committed to this leg

var float  firstEntryTimeGlobal    = na
var float  lastExitTimeGlobal      = na
var float  totalCapitalDeployed    = 0.0        // running capital in use across open legs

wasInTrade = array.size(legEntry) > 0
inTrade    = wasInTrade

watchCondition = enableWatch and not inTrade and not athSignal and not na(distanceFromATH) and
                 distanceFromATH >= 0 and distanceFromATH <= watchZonePct and consolMonths >= 2

readyCondition = enableReady and not inTrade and not athSignal and not na(distanceFromATH) and
                 distanceFromATH >= 0 and distanceFromATH <= readyZonePct and
                 isTightBase and volLooksGood and consolMonths >= 3 and compositeScoreLive >= scoreThreshold

watchTriggered = watchCondition and not watchCondition[1]
readyTriggered = readyCondition and not readyCondition[1]

// ============================================================
// POSITION SIZING — only meaningful on a confirmEntryTrigger bar.
// Entry price = triggerLevel (the armed level, "just above the
// wick"), not the confirming candle's high or close.
// ============================================================
potentialEntryP  = triggerLevel
potentialSL      = stopTypeInput == "ATR" ? potentialEntryP - atrValue * atrMult : low
potentialRisk    = potentialEntryP - potentialSL
riskAmountCcy    = accountCapital * riskPerLegPct / 100.0
potentialQty     = (confirmEntryTrigger and potentialRisk > 0) ? math.floor(riskAmountCcy / potentialRisk) : 0.0
potentialCapital = potentialQty * potentialEntryP
capitalCeiling   = accountCapital * maxCapitalDeployPct / 100.0
capitalRoomLeft  = math.max(capitalCeiling - totalCapitalDeployed, 0.0)

// ============================================================
// BUY (new leg) — fires when a later month confirms above the
// armed trigger level, gated by BOTH leg count cap AND capital cap.
// ============================================================
canAddLeg      = array.size(legEntry) < maxPyramidLegs and potentialQty > 0 and
                 (totalCapitalDeployed + potentialCapital) <= capitalCeiling
buySignal      = confirmEntryTrigger and canAddLeg
firstEntrySignal = buySignal and not wasInTrade
addSignal      = buySignal and wasInTrade
legCapBlocked  = confirmEntryTrigger and array.size(legEntry) < maxPyramidLegs and not canAddLeg   // capital blocked it specifically
legCapReached  = confirmEntryTrigger and not canAddLeg      // confirmed trigger occurred but something blocked a new leg (count OR capital)

if buySignal
    trendScoreV = f_trendScore(close, smaFast, smaSlow, higherHigh, higherLow)
    baseScoreV  = f_baseScore(tightnessPct, tightPct, consolMonths, consolHistoryArr, fallbackMed, fallbackHigh, tooLongMonths, risingLowsInBase, volDryingUp)
    volScoreV   = f_volumeScore(volume, volAvg, volMultiplier, volDryingUp)

    entryP = potentialEntryP
    slL    = potentialSL
    riskA  = potentialRisk
    t1     = entryP + riskA * rr1
    t2     = entryP + riskA * rr2
    qtyV   = potentialQty
    capV   = potentialCapital

    array.push(legEntry, entryP)
    array.push(legSL, slL)
    array.push(legT1, t1)
    array.push(legT2, t2)
    array.push(legEntryBar, bar_index)
    array.push(legEntryTime, time)
    array.push(legRisk, riskA)
    array.push(legTrend, trendScoreV)
    array.push(legBase, baseScoreV)
    array.push(legVol, volScoreV)
    array.push(legConsol, consolMonths)
    array.push(legQty, qtyV)
    array.push(legCapital, capV)

    totalCapitalDeployed := totalCapitalDeployed + capV

    array.push(consolHistoryArr, float(consolMonths))
    if na(firstEntryTimeGlobal)
        firstEntryTimeGlobal := time

sellSignal = wasInTrade and exitTrigger   // closes ALL open legs together

// ============================================================
// STATS ENGINE — per-leg records, currency P&L
// ============================================================
var float[] tradePnlPct          = array.new_float(0)
var float[] tradePnlCcy          = array.new_float(0)
var float[] tradeCloseBarIdxArr  = array.new_float(0)
var float[] tradeRMultipleArr    = array.new_float(0)
var float[] tradeDurationBarsArr = array.new_float(0)
var float[] tradeDurationDaysArr = array.new_float(0)
var float   lastBatchAvgPnl      = na
var float   lastBatchPnlCcy      = na
var int     lastBatchLegCount    = 0
var float   cumPnlCcy            = 0.0   // running net P&L in currency across all closed legs

if sellSignal
    lastBatchLegCount := array.size(legEntry)
    float batchPnlSum    = 0.0
    float batchPnlCcySum = 0.0
    for i = 0 to array.size(legEntry) - 1
        entryP = array.get(legEntry, i)
        qtyI   = array.get(legQty, i)
        pnl    = (close - entryP) / entryP * 100
        pnlCcy = (close - entryP) * qtyI
        array.push(tradePnlPct, pnl)
        array.push(tradePnlCcy, pnlCcy)
        array.push(tradeCloseBarIdxArr, float(bar_index))

        durationBars = bar_index - array.get(legEntryBar, i)
        array.push(tradeDurationBarsArr, float(durationBars))

        durationDays = (time - array.get(legEntryTime, i)) / 86400000.0
        array.push(tradeDurationDaysArr, durationDays)

        riskA = array.get(legRisk, i)
        rMultiple = riskA != 0 ? (close - entryP) / riskA : na
        array.push(tradeRMultipleArr, rMultiple)

        batchPnlSum    += pnl
        batchPnlCcySum += pnlCcy

    lastBatchAvgPnl := batchPnlSum / array.size(legEntry)
    lastBatchPnlCcy := batchPnlCcySum
    cumPnlCcy       := cumPnlCcy + batchPnlCcySum
    lastExitTimeGlobal := time
    totalCapitalDeployed := 0.0   // all legs closed together, so capital fully freed

    array.clear(legEntry)
    array.clear(legSL)
    array.clear(legT1)
    array.clear(legT2)
    array.clear(legEntryBar)
    array.clear(legEntryTime)
    array.clear(legRisk)
    array.clear(legTrend)
    array.clear(legBase)
    array.clear(legVol)
    array.clear(legConsol)
    array.clear(legQty)
    array.clear(legCapital)

totalTrades = array.size(tradePnlPct)   // = total closed LEGS across all positions

var float winCount           = 0.0
var float sumPnl             = 0.0
var float bestTrade          = na
var float worstTrade         = na
var float sumWinPnl          = 0.0
var float sumLossPnl         = 0.0
var float winCountForAvg     = 0.0
var float lossCountForAvg    = 0.0
var float avgWin             = na
var float avgLoss            = na
var float profitFactor       = na
var float maxDD              = 0.0
var float maxDDDurationBars  = 0.0
var int   maxWinStreak       = 0
var int   maxLossStreak      = 0
var float sumRMultiple       = 0.0
var int   rCount             = 0
var float avgRMultiple       = na
var float sumDurationBars    = 0.0
var float sumDurationDays    = 0.0
var float avgHoldingBars     = na
var float avgDurationDays    = na
var float cumEquityMult      = 1.0
var float totalReturnPct     = na
var float cagrPct            = na
var float expectancyPct      = na
var float avgPnl             = na

if barstate.islast and totalTrades > 0
    winCount         := 0.0
    sumPnl           := 0.0
    bestTrade        := array.get(tradePnlPct, 0)
    worstTrade       := array.get(tradePnlPct, 0)
    sumWinPnl        := 0.0
    sumLossPnl       := 0.0
    winCountForAvg   := 0.0
    lossCountForAvg  := 0.0
    maxWinStreak     := 0
    maxLossStreak    := 0
    sumRMultiple     := 0.0
    rCount           := 0
    sumDurationBars  := 0.0
    sumDurationDays  := 0.0
    cumEquityMult    := 1.0

    int curWinStreak  = 0
    int curLossStreak = 0
    cumEquity  = 0.0
    peakEquity = 0.0
    peakBarIdx = array.get(tradeCloseBarIdxArr, 0)
    maxDD             := 0.0
    maxDDDurationBars := 0.0

    for i = 0 to totalTrades - 1
        p      = array.get(tradePnlPct, i)
        barIdx = array.get(tradeCloseBarIdxArr, i)
        sumPnl += p
        cumEquityMult := cumEquityMult * (1.0 + p / 100.0)

        if p > 0
            winCount       += 1
            sumWinPnl      += p
            winCountForAvg += 1
            curWinStreak   += 1
            curLossStreak  := 0
            maxWinStreak   := math.max(maxWinStreak, curWinStreak)
        else if p < 0
            sumLossPnl      += p
            lossCountForAvg += 1
            curLossStreak   += 1
            curWinStreak    := 0
            maxLossStreak   := math.max(maxLossStreak, curLossStreak)
        else
            curWinStreak  := 0
            curLossStreak := 0

        if p > bestTrade
            bestTrade := p
        if p < worstTrade
            worstTrade := p

        rVal = array.get(tradeRMultipleArr, i)
        if not na(rVal)
            sumRMultiple += rVal
            rCount += 1

        sumDurationBars += array.get(tradeDurationBarsArr, i)
        sumDurationDays += array.get(tradeDurationDaysArr, i)

        cumEquity += p
        if cumEquity > peakEquity
            peakEquity := cumEquity
            peakBarIdx := barIdx
        dd = peakEquity - cumEquity
        if dd > maxDD
            maxDD             := dd
            maxDDDurationBars := barIdx - peakBarIdx

    if cumEquity < peakEquity
        currentDDBars = bar_index - peakBarIdx
        if currentDDBars > maxDDDurationBars
            maxDDDurationBars := currentDDBars

    avgWin          := winCountForAvg  > 0 ? sumWinPnl / winCountForAvg   : na
    avgLoss         := lossCountForAvg > 0 ? sumLossPnl / lossCountForAvg : na
    profitFactor    := (lossCountForAvg > 0 and sumLossPnl != 0) ? sumWinPnl / math.abs(sumLossPnl) : na
    avgRMultiple    := rCount > 0 ? sumRMultiple / rCount : na
    avgHoldingBars  := sumDurationBars / totalTrades
    avgDurationDays := sumDurationDays / totalTrades
    totalReturnPct  := (cumEquityMult - 1.0) * 100.0
    avgPnl          := sumPnl / totalTrades
    expectancyPct   := avgPnl

    yearsSpan = (not na(firstEntryTimeGlobal) and not na(lastExitTimeGlobal)) ? (lastExitTimeGlobal - firstEntryTimeGlobal) / (365.25 * 86400000.0) : na
    cagrPct := (not na(yearsSpan) and yearsSpan > 0) ? (math.pow(cumEquityMult, 1.0 / yearsSpan) - 1.0) * 100.0 : na

winRate = totalTrades > 0 ? winCount / totalTrades * 100 : na

openLegCount = array.size(legEntry)

statusText  = sellSignal ? "SELL (all legs)" : firstEntrySignal ? "BUY" : addSignal ? "ADD (leg " + str.tostring(openLegCount) + ")" : legCapBlocked ? "TRIGGER HIT (capital cap)" : legCapReached ? "TRIGGER HIT (cap reached)" : wasInTrade ? "IN TRADE (" + str.tostring(openLegCount) + " legs)" : readyCondition ? "READY" : watchCondition ? "WATCH" : not na(pendingLevel) ? "ARMED (awaiting confirmation)" : "IGNORE"
statusColor = sellSignal ? color.red : firstEntrySignal ? color.lime : addSignal ? color.aqua : legCapBlocked ? color.orange : legCapReached ? color.orange : wasInTrade ? color.green : readyCondition ? color.aqua : watchCondition ? color.yellow : not na(pendingLevel) ? color.fuchsia : color.silver

recCategory = firstEntrySignal ? "BUY" : addSignal ? "ADD" : legCapReached ? "IGNORE" : readyCondition ? "READY" : watchCondition ? "WATCH" : "IGNORE"

f_recommendationReason(cat, tight, volGood, capHit, capBlocked, armed) =>
    string txt = ""
    if cat == "BUY"
        txt := "Confirmed above trigger level - first entry"
    else if cat == "ADD"
        txt := "New confirmed trigger - pyramid leg added"
    else if capBlocked
        txt := "Trigger confirmed but capital cap reached - can't size a new leg"
    else if capHit
        txt := "Trigger confirmed but max legs reached"
    else if cat == "READY"
        txt := "Near ATH, tight base, good volume"
    else if cat == "WATCH"
        txt := tight ? "Near ATH, base tightening" : "Near ATH, needs more consolidation"
    else if armed
        txt := "ATH wick made - waiting for a later month to confirm above " + str.tostring(pendingLevel, "#.##")
    else
        txt := "Not near ATH / weak setup"
    txt

recReason = f_recommendationReason(recCategory, isTightBase, volLooksGood, legCapReached, legCapBlocked, not na(pendingLevel))

mostRecentIdx   = openLegCount - 1
displayTrend    = openLegCount > 0 ? array.get(legTrend, mostRecentIdx) : trendScoreLive
displayBase     = openLegCount > 0 ? array.get(legBase, mostRecentIdx)  : baseScoreLive

    table.cell(t, 1, rowIdx, value, text_color=valColor, bgcolor=color.new(color.gray, 40), text_size=size.small)
